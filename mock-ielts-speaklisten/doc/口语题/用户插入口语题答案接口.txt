接口文档：用户插入口语题答案接口

1. 接口名称：用户提交口语题答案
   请求方式：POST
   URL：http://localhost:9486/speak/userAnswer/submit
   请求参数：
       - Body (JSON格式)：
         {
             "SpeakUserAnswerRecord": {
                 "recordId": Long,          // 答题记录ID（可选，新增时不传）
                 "userId": Integer,         // 用户ID
                 "speakingId": Long,        // 口语材料ID
                 "score": String,           // 答题得分（可选）
                 "answerEvaluation": String,// 答题评价（可选）
                 "durationSeconds": Integer,// 答题耗时（秒）
                 "deviceType": String,      // 设备类型（WEB, MOBILE, TABLET）
                 "part": String             // 题目所属部分（Part1, Part2, Part3）
             },
             "parts": {
                 "Part1": [
                     {
                         "answer": {
                             "detailId": Long,          // 答案明细ID（可选，新增时不传）
                             "recordId": Long,          // 答题记录ID（可选）
                             "userId": Integer,         // 用户ID
                             "questionId": Long,        // 题目ID
                             "userAudioUrl": String,    // 用户音频URL
                             "userTranscript": String,  // 用户转录文本
                             "score": String,           // 答案得分（可选）
                             "feedback": String,        // 答案反馈（可选）
                             "part": String,             // 题目所属部分（Part1, Part2, Part3）
                             "questionContent": String, // 题目内容
                         }
                     }
                 ]
             }
         }
   响应结果：
       - 成功：
         {
             "code": 200,
             "message": "用户口语题答案提交成功",
             "data": null
         }
       - 失败：
         {
             "code": 400,
             "message": "错误信息",
             "data": null
         }

2. 接口名称：根据记录ID获取用户口语答题记录
   请求方式：GET
   URL：/speak/userAnswer/{recordId}
   请求参数：
       - Path Variable：
         recordId: Long (必填) 答题记录ID
   响应结果：
       - 成功：
         {
             "code": 200,
             "message": "获取用户口语答题记录成功",
             "data": {
                 "SpeakUserAnswerRecord": {
                     "recordId": Long,
                     "userId": Integer,
                     "speakingId": Long,
                     "score": String,
                     "answerEvaluation": String,
                     "durationSeconds": Integer,
                     "deviceType": String,
                     "part": String
                 },
                 "parts": {
                     "Part1": [
                         {
                             "answer": {
                                 "detailId": Long,
                                 "recordId": Long,
                                 "userId": Integer,
                                 "questionId": Long,
                                 "userAudioUrl": String,
                                 "userTranscript": String,
                                 "score": String,
                                 "feedback": String,
                                 "part": String
                             }
                         }
                     ]
                 }
             }
         }
       - 失败：
         {
             "code": 400,
             "message": "错误信息",
             "data": null
         }
